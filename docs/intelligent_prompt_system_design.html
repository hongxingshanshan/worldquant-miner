市值标准化 (divide(field, cap)) 0% 100% 必须添加

3.4 文本向量化策略

Alpha 表达式的向量化需要包含语义信息，建议组合以下内容：

def build_alpha_document(alpha_data: dict) -> str:
    """构建 Alpha 文档用于向量化"""
    parts = []

    # 1. 表达式本身
    parts.append(f"表达式: {alpha_data['expression']}")

    # 2. 结构特征描述（基于实际分析的成功模式）
    features = []

    # 外层操作符类型（关键特征）
    outer_op = alpha_data.get('outer_operator', 'none')
    if outer_op != 'none':
        features.append(f"外层使用 {outer_op}() 进行权重分布")
    else:
        features.append("缺少外层排名操作符")

    # 市值标准化（100% 成功 Alpha 都有）
    if alpha_data.get('has_divide_cap'):
        features.append("使用市值标准化 divide(field, cap)")

    # 行业中性化（35% 成功 Alpha 有）
    if alpha_data.get('has_group_neutralize'):
        features.append("使用行业中性化")

    # 时间窗口
    if alpha_data.get('time_window'):
        features.append(f"时间窗口 {alpha_data['time_window']} 天")

    parts.append("特征: " + ", ".join(features))

    # 3. 性能指标描述
    sharpe = alpha_data.get('sharpe', 0) or 0
    fitness = alpha_data.get('fitness', 0) or 0
    turnover = alpha_data.get('turnover', 0) or 0
    parts.append(f"Sharpe: {sharpe:.2f}, Fitness: {fitness:.2f}, Turnover: {turnover:.2f}")

    # 4. 数据字段描述
    if alpha_data.get('data_fields'):
        parts.append(f"数据字段: {', '.join(alpha_data['data_fields'][:5])}")

    # 5. 失败原因（如果有）
    if alpha_data.get('fail_type'):
        parts.append(f"失败类型: {alpha_data['fail_type']}")
        if alpha_data.get('fail_check_name'):
            parts.append(f"失败检查: {alpha_data['fail_check_name']}")

    return "\n".join(parts)


def extract_outer_operator(expression: str) -> str:
    """提取最外层操作符"""
    expr = expression.strip()

    # 检查最外层操作符
    if expr.startswith('rank('):
        return 'rank'
    elif expr.startswith('ts_rank('):
        return 'ts_rank'
    elif expr.startswith('group_rank('):
        return 'group_rank'
    elif expr.startswith('group_neutralize('):
        # group_neutralize 内部可能有 rank
        return 'group_neutralize'
    else:
        return 'none'


def extract_time_window(expression: str) -> Optional[int]:
    """提取时间窗口参数"""
    import re
    # 匹配 ts_xxx(expr, N) 或 ts_xxx(expr, N, M) 中的 N
    matches = re.findall(r'ts_\w+\([^,]+,\s*(\d+)', expression)
    if matches:
        return int(matches[0])
    return None

# generation_one/naive-ollama/prompt_builder.py

class IntelligentPromptBuilder:
    """智能提示词构建器"""

    def __init__(self, retriever: KnowledgeRetriever):
        self.retriever = retriever

    def build_generation_prompt(
        self,
        data_fields: List[str],
        operators: List[dict],
        context: dict = None
    ) -> str:
        """构建 Alpha 生成提示词"""

        # 1. 检索成功案例
        success_cases = self.retriever.retrieve(
            query="高 Sharpe 高 Fitness 的成功 Alpha 表达式模式",
            collection="alpha_submitted",
            top_k=5
        )

        # 2. 检索失败模式（用于避免）
        failure_patterns = self.retriever.retrieve(
            query="LOW_FITNESS 失败的 Alpha 特征",
            collection="alpha_failure",
            top_k=3
        )

        # 3. 检索字段组合建议
        field_combos = self.retriever.retrieve(
            query=f"数据字段 {' '.join(data_fields[:10])} 的成功组合",
            collection="field_combinations",
            top_k=3
        )

        # 4. 组装提示词
        return self._assemble_prompt(
            data_fields=data_fields,
            operators=operators,
            success_cases=success_cases,
            failure_patterns=failure_patterns,
            field_combos=field_combos,
            context=context
        )

    def build_optimization_prompt(
        self,
        alpha_data: dict,
        fail_type: str
    ) -> str:
        """构建 Alpha 优化提示词"""

        # 1. 检索同类失败的成功优化案例
        optimization_cases = self.retriever.retrieve(
            query=f"{fail_type} 失败优化成功的案例",
            collection="alpha_optimization_cases",
            top_k=5
        )

        # 2. 检索相似表达式的成功版本
        similar_success = self.retriever.retrieve(
            query=alpha_data['expression'],
            collection="alpha_submitted",
            top_k=3
        )

        # 3. 检索针对性修复知识
        fix_knowledge = self.retriever.retrieve(
            query=f"如何修复 {fail_type} 检查失败",
            collection="knowledge_chunks",
            category_filter="technical_methods",
            top_k=3
        )

        return self._assemble_optimization_prompt(
            alpha_data=alpha_data,
            fail_type=fail_type,
            optimization_cases=optimization_cases,
            similar_success=similar_success,
            fix_knowledge=fix_knowledge
        )

# scripts/sync_alpha_vectors.py

"""
Alpha 向量同步启动脚本
定时将 MySQL 中的 Alpha 数据同步到向量数据库
"""

import schedule
import time
from vector_store.alpha_vector_sync import AlphaVectorSync
from db.db_connector import DBConnector
import json

def load_config():
    with open('db/db_config.json') as f:
        return json.load(f)

def sync_job():
    """同步任务"""
    db = DBConnector(load_config())
    sync = AlphaVectorSync(db)

    print("开始同步已提交 Alpha...")
    submitted_count = sync.sync_submitted_alphas()
    print(f"同步已提交 Alpha: {submitted_count}")

    print("开始同步可提交 Alpha...")
    submittable_count = sync.sync_submittable_alphas()
    print(f"同步可提交 Alpha: {submittable_count}")

    print("开始同步失败 Alpha...")
    failure_count = sync.sync_failure_alphas()
    print(f"同步失败 Alpha: {failure_count}")

    print("同步完成!")

def main():
    # 立即执行一次
    sync_job()

    # 每小时执行增量同步
    schedule.every(1).hours.do(sync_job)

    # 每天凌晨 3 点全量同步
    schedule.every().day.at("03:00").do(lambda: AlphaVectorSync(DBConnector(load_config())).full_sync())

    while True:
        schedule.run_pending()
        time.sleep(60)

if __name__ == "__main__":
    main()

8.5 权重动态调整机制

# prompt_builder.py 中的权重管理

class PromptWeightManager:
    """提示词权重动态管理"""

    def __init__(self):
        self.phase = "short"  # short / mid / long
        self.weights = {
            "short": {"expert": 0.7, "retrieval": 0.3},
            "mid": {"expert": 0.4, "retrieval": 0.6},
            "long": {"expert": 0.2, "retrieval": 0.8}
        }

    def get_current_weights(self) -> dict:
        """获取当前权重配置"""
        return self.weights[self.phase]

    def check_phase_transition(self, stats: dict) -> str:
        """检查是否需要阶段转换"""
        # 中期条件：知识块 > 1000 或 外部知识 > 200
        if self.phase == "short":
            if stats["total_chunks"] > 1000 or stats["external_knowledge"] > 200:
                self.phase = "mid"
                return "transition_to_mid"

        # 长期条件：知识块 > 5000 且 成功案例 > 500
        if self.phase == "mid":
            if stats["total_chunks"] > 5000 and stats["success_cases"] > 500:
                self.phase = "long"
                return "transition_to_long"

        return "no_transition"

    def build_prompt(self, retrieved_context: str, expert_knowledge: str) -> str:
        """根据权重构建提示词"""
        weights = self.get_current_weights()

        if weights["retrieval"] >= weights["expert"]:
            # 检索优先模式
            return LONG_TERM_PROMPT_TEMPLATE.format(
                retrieved_context=retrieved_context,
                expert_knowledge=expert_knowledge,
                ...
            )
        else:
            # 专家优先模式
            return SHORT_TERM_PROMPT_TEMPLATE.format(
                expert_role=EXPERT_ROLE_PROMPT,
                retrieved_context=retrieved_context,
                ...
            )
